CODE : Little Frog
```cpp
#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>> n;
    int i=1;
    int j=n;
    while(i<j){
        cout<<i<<" ";
        i++;
        cout<<j<<" ";
        j--;
    }
    if(i==j){
        cout<<i;
    }
}
```
